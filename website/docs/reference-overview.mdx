import React from 'react'
import Intro from '@site/src/components/Intro'
import DocCardList from '@theme/DocCardList'
import {useDocsSidebar} from '@docusaurus/plugin-content-docs/client'

export function ReferenceSections() {
  const sidebar = useDocsSidebar();
  const groups = [];
  for (const item of sidebar.items) {
    const label = item.customProps?.navigationGroup;
    if (label) groups.push({label, items: []});
    else if (!item.customProps?.navigationOverview && groups.length) {
      groups[groups.length - 1].items.push(item);
    }
  }
  return groups.map(({label, items}) => (
    <section key={label}>
      <h2>{label}</h2>
      <DocCardList items={items} />
    </section>
  ));
}

<Intro>
Find the commands and configuration you need to work with Atmos. Start with the CLI essentials, build a CI/CD workflow, or explore the components, stacks, and automation that organize your infrastructure.
</Intro>
